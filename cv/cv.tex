% ================= RESEARCH PROJECTS =================
\section{Research Projects}
\entry{Safe Inference-Time Alignment (LARA)}{SCALAR Lab, UMass Amherst}{COLM 2026}{}
\begin{itemize}
  \item Moved the Safe RLHF constraint \textbf{from training to decoding}: dualizing the constraint reduces the search over policies to a
        \textbf{one-dimensional convex problem} over a \textbf{single Lagrange multiplier} $\lambda^\star$.
  \item Calibrated $\lambda^\star$ by binary search on a small prompt set in \hl{about two minutes}; the resulting reward $r - \lambda^\star c$
        plugs into Best-of-$N$, ARGS, PARGS, or CD-Q with \textbf{no weight updates}, and Best-of-$N$ \textbf{approaches finetuned baselines}.
\end{itemize}
\vspace{3pt}

\entry{High-Confidence Safe RLHF (HC-RLHF)}{SCALAR Lab, UMass Amherst}{RLC 2025}{}
\begin{itemize}
  \item Proposed the \textbf{first Seldonian algorithm for RLHF}: optimize helpfulness under a pessimistic upper-confidence bound on cost,
        then return the model only if it passes a \textbf{held-out statistical safety test}.
  \item Derived a \textbf{variance-aware policy gradient} for candidate selection, \textbf{proved} that an unsafe model is returned with probability
        \hl{at most $\delta$}, and validated on Qwen2-1.5B, Qwen2.5-3B, and Llama-3.2-3B with improved harmlessness and helpfulness.
\end{itemize}
\vspace{3pt}

\entry{Risk-Sensitive Alignment beyond Expected Cost}{SCALAR Lab, UMass Amherst}{Preprint 2026}{}
\begin{itemize}
  \item Replaced the expected-cost constraint in Safe RLHF with \textbf{first-order stochastic dominance} over the full cost distribution,
        which controls a broad family of \textbf{spectral risk measures}; made it tractable with optimal transport and Sinkhorn iterations.
\end{itemize}
\vspace{3pt}

\entry{Adaptive Margin RLHF via Preference over Preferences}{SCALAR Lab, UMass Amherst}{Preprint 2025}{}
\begin{itemize}
  \item Proposed \textbf{DPO-PoP}, which uses preference-over-preference supervision to infer \textbf{adaptive margins}, \textbf{outperforming fixed and
        ground-truth margin baselines} on UltraFeedback; analyzed the trade-off between discriminative and generative performance
        and designed sampling strategies to prioritize either.
\end{itemize}
\vspace{3pt}

\entry{Reward Overoptimization in Direct Alignment Algorithms}{SCALAR Lab, UMass Amherst}{NeurIPS 2024}{}
\begin{itemize}
  \item Showed that direct alignment algorithms (DPO, IPO, SLiC) \textbf{degrade at high KL budgets} like classical RLHF, \textbf{begin overfitting
        before a single epoch completes}, and, being under-constrained, often \textbf{favor out-of-distribution responses}; smaller models
        overfit to simple features such as length.
\end{itemize}
\vspace{3pt}

\entry{Model-Free RL for Diabetes Treatment}{UMass Amherst}{M.S. project}{}
\begin{itemize}
  \item Learned \textbf{insulin-dosing policies} from continuous glucose monitoring data with DDPG, using \textbf{high-confidence off-policy
        improvement} to switch from the behavior policy only when the new policy is better with high confidence.
\end{itemize}
